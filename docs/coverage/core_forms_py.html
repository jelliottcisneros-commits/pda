98class ConsultantCreationForm(ModelForm): 

99 """Form used by admin for adding new consultants""" 

100 

101 class Meta: 

102 model = Consultant 

103 fields = ('email',) 

104 field_classes = {'email': EmailField} 

105 

106 @staticmethod 

107 def generate_unique_username_from_email(email): 

108 def is_username_unique_among_consultants(username): 

109 return AuthUser.objects.filter(username=username).count() == 0 

110 

111 username = email.split('@')[0] 

112 validator = UnicodeUsernameValidator() 

113 try: 

114 validator(username) # ensure the username is valid unicode 

115 except ValidationError: 

116 username = get_random_string() # if not generate a random one that is valid 

117 while not is_username_unique_among_consultants(username): 

118 username += get_random_string( 

119 length=1) # trying to make it unique while not going too far from the original 

120 return username 

121 

122 def save(self, commit=True): 

123 # Overriding so that admin only has to provide an email, and all the other fields will be generated 

124 self.instance.username = self.generate_unique_username_from_email( 

125 self.cleaned_data['email']) # setting username from the user in the email 

126 self.instance.is_staff = True 

127 consultant = super().save(commit=False) 

128 consultant.set_password(BaseUserManager().make_random_password()) # setting random password, 

129 # consultant will get to reset later. 

130 if commit: 

131 consultant.save() 

132 return consultant
