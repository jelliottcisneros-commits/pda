146def is_function_wrapped_by_decorator(function: callable, decorator_name: str): 

147 """ 

148 modified from https://schinckel.net/2012/01/20/get-decorators-wrapping-a-function/ 

149 :param function: The function in question 

150 :param decorator_name: The name of the decorator 

151 :return: True if the decorator is present, False otherwise 

152 """ 

153 

154 def get_callable_cells(func): 

155 callables = [] 

156 if not hasattr(func, '__closure__'): 

157 if hasattr(func, 'view_func'): 

158 return get_callable_cells(func.view_func) 

159 return [] 

160 if not func.__closure__: 

161 return [func] 

162 for closure in func.__closure__: 

163 contents = closure.cell_contents 

164 if isinstance(contents, list): 

165 for content in contents: 

166 callables.extend(get_callable_cells(content)) 

167 else: 

168 callables.extend(get_callable_cells(contents)) 

169 return [func] + callables 

170 

171 callable_cells = get_callable_cells(function) 

172 for i in range(len(callable_cells)): 

173 cell = callable_cells[i] 

174 if decorator_name in str(cell): 

175 return True 

176 return False
